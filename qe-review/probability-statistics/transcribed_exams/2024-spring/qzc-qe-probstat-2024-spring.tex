\section*{Instructions}
\begin{itemize}
    \item There are 11 problems in this exam. You choose 8 of them to finish. If you select more than 8, only the first 8 that you have worked on will be graded. Note that 4 of the problems are worth 15 points each and the rest 10 points each.
    \item You must follow all the rules of exam taking. Misconducts will be subject to proper disciplinary actions by the Center.
    \item You must provide all necessary details for full credits. A final answer with no or little explanation/derivation, even if correct, receives a minimal credit.
    \item Note: $\ln x \coloneqq \log_e x$. $\mathbb{R}$ represents the set of all real numbers. $\mathbb{N} \coloneqq \{1, 2, 3, \dots\}$ denotes the set of all positive integers.
\end{itemize}

\noindent\rule{\linewidth}{0.4pt}

\begin{enumerate}[label=\textbf{\arabic*.}]
    \item \textbf{(10 points)} Let $X_1, X_2, \dots$ be independent identically distributed random variables with density
    \[
    f(x) = \begin{cases} |x|^{-3}, & |x| > 1, \\ 0, & \text{otherwise}, \end{cases}
    \]
    and characteristic function $\phi(t) \coloneqq \E e^{i t X_1}$. Let $S_n \coloneqq X_1 + \dots + X_n$ for each $n \in \mathbb{N}$.
    \begin{enumerate}[label=(\alph*)]
        \item Prove that $\E(X_1^2) = \infty$.
        \item It is known that $\lim_{t \to 0} \frac{1 - \phi(t)}{t^2 \ln |t|} = -1$. Using this fact, find a sequence $(a_n)_{n \in \mathbb{N}}$ and prove that $S_n / a_n$ converges in distribution to the standard normal distribution $\mathcal{N}(0, 1)$.
    \end{enumerate}

    \item \textbf{(15 points)} Let $X_1, X_2, \dots$ be i.i.d.\ random variables with exponential distribution: $\Pbb(X_k > x) = e^{-x}$ for $x \ge 0$. Let $M_n \coloneqq \max_{1 \le k \le n} X_k$. Show that:
    \begin{enumerate}[label=(\alph*)]
        \item $\limsup_{n \to \infty} X_n / \ln n = 1$ almost surely.
        \item $\lim_{n \to \infty} M_n / \ln n = 1$ almost surely.
    \end{enumerate}

    \item \textbf{(10 points)}
    \begin{enumerate}[label=(\alph*)]
        \item Prove that: if $X_n \xrightarrow{\Pbb} X$ (i.e., $X_n$ converges to $X$ in probability) and $Y_n \xrightarrow{\Pbb} Y$, then $X_n Y_n \xrightarrow{\Pbb} XY$.
        \item Is the following true or false: if $X_n \xrightarrow{L^1} X$ and $Y_n \xrightarrow{L^1} Y$, then $X_n Y_n \xrightarrow{L^1} XY$? Either prove it or give a counterexample.
    \end{enumerate}

    \item \textbf{(15 points)} Let $p \in (0, 1)$ and $k \in \mathbb{N}$. Let $X_1, X_2, \dots$ be independent random variables with $\Pbb(X_n = 1) = p = 1 - \Pbb(X_n = -1)$ for each $n \in \mathbb{N}$. Let $X_0 \coloneqq 0$ and $S_n \coloneqq X_0 + X_1 + \dots + X_n$ for $n \in \mathbb{N} \cup \{0\}$. Let $T_k \coloneqq \inf \{n \in \mathbb{N} : S_n = k\}$ with $\inf \varnothing = \infty$. For $t \in \mathbb{R}$, let $M(t) \coloneqq \E e^{t X_1}$.
    \begin{enumerate}[label=(\alph*)]
        \item If $p \in [1/2, 1)$, prove that $e^{t k} \E [M(t)^{-T_k}] = 1$ for all $t > 0$.
        \item If $p \in (0, 1/2)$, compute $\Pbb(T_k < \infty)$.
        \item If $p \in (0, 1/2)$, find the distribution of $Y \coloneqq 1 + \sup_{n > 0} S_n$.
    \end{enumerate}

    \item \textbf{(10 points)} Let $S$ be a countable state space and $(Z_n, n \in \mathbb{N})$ be a sequence of i.i.d.\ random variables taking values in a measurable space $(E, \mathcal{E})$.
    \begin{enumerate}[label=(\alph*)]
        \item Suppose that $f : S \times E \to S$ is a measurable mapping and $X_0$ is independent of all $Z_n$. Let $X_n \coloneqq f(X_{n-1}, Z_n), n \in \mathbb{N}$. Prove that $(X_n, n \in \mathbb{N} \cup \{0\})$ is a Markov chain, and determine its transition probability.
        \item Prove that any (discrete) time-homogeneous Markov chain on $S$ can be written in the form above.
    \end{enumerate}

    \item \textbf{(10 points)} Let $(B_t, t \ge 0)$ be a 1D Brownian motion starting from $0$. Let $\mathcal{F}_t \coloneqq \sigma(B_s : s \le t)$ be the filtration generated by $B_t$. For $a > 0$, define $T_a \coloneqq \inf \{t > 0 : B_t \ge a\}$.
    \begin{enumerate}[label=(\alph*)]
        \item Prove that $T_a$ is an $\mathcal{F}_t$ stopping time.
        \item Compute $\E e^{-\lambda T_a}$ for $\lambda > 0$, and prove that $\Pbb(T_a < \infty) = 1$. (\textit{Hint:} Consider $M_t \coloneqq e^{u B_t - u^2 t / 2}$.)
    \end{enumerate}

    \item \textbf{(15 points)} Let $X_i \sim \mathcal{N}(\mu_i, 1)$ independently for $i = 1, \dots, n$. Let $\bm{\mu} = (\mu_1, \dots, \mu_n)^T$, $\mathbf{X} = (X_1, \dots, X_n)^T$, $\mathbf{a} = (a_1, \dots, a_n)^T$. Consider the loss function $l(\bm{\mu}, \mathbf{a}) = \sum_{i=1}^n (a_i - \mu_i)^2$. Prove that $\delta(\mathbf{X}) = \mathbf{X}$ is minimax.

    \item \textbf{(10 points)} Suppose $X_1, \dots, X_n$ are i.i.d.\ Uniform$(\theta, \theta + 10)$ with $\theta > 0$.
    \begin{enumerate}[label=(\alph*)]
        \item Prove that $\hat{\theta} = \min(X_1, \dots, X_n)$ is a consistent estimator of $\theta$.
        \item Find the asymptotic distribution of $\sqrt{12n} \left(\frac{\overline{X} - 5 - \theta}{\hat{\theta}}\right)$, where $\overline{X}$ is the sample mean.
    \end{enumerate}

    \item \textbf{(10 points)} Let $(X_1, \dots, X_n)$ be a random sample where $\log X_i \sim \mathcal{N}(\theta, \theta)$ with unknown $\theta > 0$.
    \begin{enumerate}[label=(\alph*)]
        \item Derive the MLE of $\theta$, denoted by $\hat{\theta}$.
        \item What is the asymptotic distribution of $\hat{\theta}$?
    \end{enumerate}

    \item \textbf{(10 points)} Let $X_1, \dots, X_n$ be incomes of $n$ persons with Pareto density $f(x \mid \theta) = 100^\theta \theta x^{-(1+\theta)}$ for $x > 100$, where $\theta > 1$. Let $\mu = \E(X)$ be the mean income. Write down the rejection region of the UMP level $\alpha$ ($0 < \alpha < 1$) test of $H_0 : \mu = \mu_0$ versus $H_1 : \mu > \mu_0$.

    \item \textbf{(15 points)} Let $\theta_1, \theta_2, \theta_3 \ge 0$ with constraint $\theta_1 + \theta_2 + \theta_3 = 1$. We observe $X_{i1} = \theta_1 + \epsilon_{i1}, X_{i2} = \theta_2 + \epsilon_{i2}, X_{i3} = \theta_3 + \epsilon_{i3}$ for $i = 1, \dots, n$, where $\epsilon_{ik} \sim \mathcal{N}(0, 1)$ are independent. Derive the uniformly minimum-variance unbiased estimator (UMVUE) for $\theta_1$.
\end{enumerate}
