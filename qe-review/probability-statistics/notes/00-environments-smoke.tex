\section{Environments}

\begin{definition}[Sample space]\label{def:sample-space}
Let $(\Omega,\F,\Pbb)$ be a probability space.
\end{definition}

\begin{theorem}[Markov inequality]\label{thm:markov}
If $X\ge 0$ and $a>0$, then $\Pbb(X\ge a)\le \E[X]/a$.
\end{theorem}

\begin{lemma}
The same bound holds after replacing $X$ by $X^p$ for $p>0$.
\end{lemma}

\begin{proposition}
Chebyshev follows from Markov applied to $(X-\E X)^2$.
\end{proposition}

\begin{corollary}
If $\Var(X)<\infty$, then $X_n\to X$ in probability whenever $\E[(X_n-X)^2]\to 0$.
\end{corollary}

\begin{example}
For $X\sim\mathrm{Exp}(1)$, Markov gives $\Pbb(X\ge t)\le e^{-t}$ only after using the mgf form; the plain first-moment bound is weaker.
\end{example}
