\section{Statement of the problems}

\examstatement{Problem 1. [15 points]}
Suppose \(k \leqslant n \leqslant m\), and \(A \in \mathbb{C}^{m \times n}\).
\begin{itemize}
    \item find a rank-\(k\) matrix \(X\) satsifying \(\|A - X\|_{2} \leqslant \|A - B\|_{2}\) for any rank-\(k\) matrix \(B\). Here \(\|\cdot\|_{2}\) is the spectral norm.
    \item find a rank-\(k\) matrix \(X\) satsifying \(\|A - X\|_{F} \leqslant \|A - B\|_{F}\) for any rank-\(k\) matrix \(B\). Here \(\|\cdot\|_{F}\) is the Frobenius norm.
\end{itemize}

\examstatement{Problem 2. [20 points]}
Consider the Gauss--Seidel iteration for solving the linear equation \(Ax = b\) for \(A = [a_{ij}] \in \mathbb{R}^{n \times n}\), \(b = [b_{i}] \in \mathbb{R}^{n}\) with \(a_{ii} \neq 0\) for \(i = 1, \dots, n\).
\begin{enumerate}[label=\arabic*:,leftmargin=2em]
    \item determine \(x^{(0)} \in \mathbb{R}^{n}\).
    \item for \(k = 0, 1, 2, \dots\) do
    \item \(x_{i}^{(k+1)} = \dfrac{1}{a_{ii}} \bigl( b_{i} - \sum_{j=1}^{i-1} a_{ij} x_{j}^{(k+1)} - \sum_{j=i+1}^{n} a_{ij} x_{j}^{(k)} \bigr)\) for \(i = 1, 2, \dots, n\).
    \item end for
\end{enumerate}
\begin{itemize}
    \item Write \(x^{(k)} = [x_{i}^{(k)}]\) and \(A = D - L - U\) where \(D, -L, -U\) are the diagonal, strictly lower triangular, strictly upper triangular parts of \(A\) respectively. Show that \(x^{(k+1)} = (D - L)^{-1} U x^{(k)} + (D - L)^{-1} b\).
    \item Prove that if \(A\) is diagonally dominant, then \(A\) is invertible, and the Gauss--Seidel iteration converges, namely \(\lim_{k \to \infty} x^{(k)} = A^{-1} b\) for any \(x^{(0)}\). (A matrix \(A\) is called diagonally dominant, if \(|a_{ii}| > \sum_{j \neq i} |a_{ij}|\) for \(i = 1, \dots, n\).)
    \item Prove that if \(A\) is symmetric and positive definite, then \(A\) is invertible, and the Gauss--Seidel iteration converges.
\end{itemize}

\examstatement{Problem 3. [20 points]}
Given the Hilbert matrix \(H_{n} = \bigl[\frac{1}{i+j-1}\bigr]_{i,j=1,\dots,n}\). Prove:
\begin{itemize}
    \item \(H_{n}\) is positive definite.
    \item the spectral radius \(\rho(H_{n})\) of \(H_{n}\) is strictly monotonically increasing with respect to \(n\).
    \item \(\rho(H_{n}) \to \pi\) as \(n \to \infty\).
\end{itemize}

\examstatement{Problem 4. [10 points]}
Apply the following three-step method with a parameter \(\theta \geqslant 0\) for solving the heat equation \(u_{t} - a^{2} u_{xx} = 0\) (Cauchy or periodic problem):

\[
(1 + \theta) \frac{u_{j}^{n+1} - u_{j}^{n}}{k} - \theta \frac{u_{j}^{n} - u_{j}^{n-1}}{k} = a^{2} \frac{u_{j+1}^{n+1} - 2u_{j}^{n+1} + u_{j-1}^{n+1}}{h^{2}}.
\]

Give its truncation error and stability. Particularly, give the value of \(\theta\) such that the truncation error attains its highest order.

\examstatement{Problem 5. [15 points]}
Denote the grid \(I_{h} = \{x_{j}\}_{j=0}^{M}\), \(x_{0} < x_{1} < \dots < x_{M}\). Set \(u = \{u_{j}\}_{j=0}^{M}\) as a grid function on \(I_{h}\). Suppose

\[
L u_{j} = -(a_{j} u_{j-1} - b_{j} u_{j} + c_{j} u_{j+1}) + q_{j} u_{j}, \qquad j = 1, \dots, M-1,
\]

where \(a_{j}, b_{j}, c_{j} > 0\), \(q_{j} \geqslant 0\) and \(a_{j} + c_{j} \leqslant b_{j}\).
\begin{itemize}
    \item Assume \(L u_{j} \leqslant 0\) for all \(1 \leqslant j \leqslant M-1\). Show that \(u_{j}\) can't attain positive maximum at inner points (\(1 \leqslant j \leqslant M-1\)) unless \(u_{j} \equiv C\);
    \item Suppose \(d_{j} = b_{j} - a_{j} - c_{j} + q_{j} > 0\) (\(j = 1, \dots, M-1\)). Show the solution of the difference equation
    \[
    L u_{j} = \varphi_{j}, \quad j = 1, \dots, M-1; \qquad u_{0} = u_{M} = 0,
    \]
    satisfies \(\|u\|_{\infty} = \max_{j} |u_{j}| \leqslant \max_{j} \dfrac{|\varphi_{j}|}{d_{j}}\).
\end{itemize}

\examstatement{Problem 6. [10 points]}
Let \(S \subset \mathbb{R}^{n}\) be a non-empty closed convex set, \(f \in C^{2}(S)\) is convex. For any \(x^{0} \in S\), define \(L(f(x^{0})) = \{x \in S \mid f(x) \le f(x^{0})\}\), if there exists some \(m > 0\) such that

\[
d^{\top} \nabla^{2} f(x) d \ge m \|d\|^{2}, \qquad \forall x \in L(f(x^{0})),\ d \in \mathbb{R}^{n}.
\]

Prove that \(L(f(x^{0}))\) is a bounded and closed convex set.

\examstatement{Problem 7. [10 points]}
Given a linear programming (LP):
\begin{align*}
\max\quad & 3x_{1} + 2x_{2} + 15x_{3} \\
\text{s.t.}\quad & x_{1} + x_{2} + x_{3} \le 12, \\
& 2x_{1} + x_{2} + 5x_{3} \le 18, \\
& x_{1}, x_{2}, x_{3} \ge 0.
\end{align*}
\begin{itemize}
    \item Write down the standard LP form.
    \item Compute the optimal solution and optimal value of the problem.
\end{itemize}
