\begin{enumerate}[label=\textbf{(\arabic*)}]
    \item \textbf{(15 points)} Suppose $k \le n \le m$, and $A \in \mathbb{C}^{m \times n}$.
    \begin{itemize}
        \item Find a rank-$k$ matrix $X$ satisfying $\|A - X\|_2 \le \|A - B\|_2$ for any rank-$k$ matrix $B$.
        \item Find a rank-$k$ matrix $X$ satisfying $\|A - X\|_F \le \|A - B\|_F$ for any rank-$k$ matrix $B$.
    \end{itemize}

    \item \textbf{(20 points)} Consider Gauss-Seidel iteration for $Ax = b$ ($a_{ii} \neq 0$):
    \[
    x_i^{(k+1)} = \frac{1}{a_{ii}} \left( b_i - \sum_{j=1}^{i-1} a_{ij} x_j^{(k+1)} - \sum_{j=i+1}^n a_{ij} x_j^{(k)} \right).
    \]
    \begin{itemize}
        \item Show that $x^{(k+1)} = (D - L)^{-1} U x^{(k)} + (D - L)^{-1} b$, where $A = D - L - U$.
        \item Prove that if $A$ is strictly diagonally dominant, Gauss-Seidel iteration converges to $A^{-1}b$.
        \item Prove that if $A$ is symmetric and positive definite, Gauss-Seidel iteration converges to $A^{-1}b$.
    \end{itemize}

    \item \textbf{(20 points)} Given the Hilbert matrix $H_n = \left[\frac{1}{i+j-1}\right]_{i,j=1,\dots,n}$. Prove:
    \begin{itemize}
        \item $H_n$ is positive definite.
        \item The spectral radius $\rho(H_n)$ is strictly monotonically increasing with respect to $n$.
        \item $\rho(H_n) \to \pi$ as $n \to \infty$.
    \end{itemize}

    \item \textbf{(10 points)} Apply the three-step method ($\theta \ge 0$) for $u_t - a^2 u_{xx} = 0$:
    \[
    (1 + \theta) \frac{u_j^{n+1} - u_j^n}{k} - \theta \frac{u_j^n - u_j^{n-1}}{k} = a^2 \frac{u_{j+1}^{n+1} - 2u_j^{n+1} + u_{j-1}^{n+1}}{h^2}.
    \]
    Give its truncation error and stability. Find $\theta$ giving highest order truncation error.

    \item \textbf{(15 points)} Grid $I_h = \{x_j\}_{j=0}^M$. $L u_j = -(a_j u_{j-1} - b_j u_j + c_j u_{j+1}) + q_j u_j$ ($a_j, b_j, c_j > 0, q_j \ge 0, a_j + c_j \le b_j$).
    \begin{itemize}
        \item Assume $L u_j \le 0$. Show $u_j$ cannot attain a positive maximum at interior points unless constant.
        \item Suppose $d_j = b_j - a_j - c_j + q_j > 0$. Show $\|u\|_\infty \le \max_j \frac{|\varphi_j|}{d_j}$ for $L u_j = \varphi_j, u_0 = u_M = 0$.
    \end{itemize}

    \item \textbf{(10 points)} Let $S \subset \mathbb{R}^n$ be closed, convex. $f \in C^2(S)$ convex. $L(f(x^0)) = \{x \in S \mid f(x) \le f(x^0)\}$. If $d^T \nabla^2 f(x) d \ge m \|d\|^2$ ($\forall x \in L(f(x^0)), m > 0$), prove $L(f(x^0))$ is bounded and closed.

    \item \textbf{(10 points)} LP: $\max 3x_1 + 2x_2 + 15x_3$ s.t. $x_1 + x_2 + x_3 \le 12$, $2x_1 + x_2 + 5x_3 \le 18$, $x_1, x_2, x_3 \ge 0$.
    \begin{itemize}
        \item Write standard form.
        \item Compute optimal solution and value.
    \end{itemize}
\end{enumerate}
