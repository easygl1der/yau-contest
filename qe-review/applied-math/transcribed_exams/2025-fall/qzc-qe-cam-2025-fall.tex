\begin{enumerate}[label=\textbf{\arabic*.}]
    \item \textbf{(10 points)} Matrix inversion iteration $X_{k+1} = 2X_k - X_k A X_k$. Prove:
    \begin{enumerate}[label=(\alph*)]
        \item If $\|I - A X_0\| < 1$, then $A$ is nonsingular and $X_k \to A^{-1}$ quadratically.
        \item There exists $X_0$ with $\|I - A X_0\| < 1$ iff $A$ is nonsingular.
    \end{enumerate}

    \item \textbf{(20 points)} $\|f\|_\infty = \max_{[-1, 1]} |f(x)|$. Chebyshev polynomial $T_n(x)$.
    \begin{enumerate}[label=(\alph*)]
        \item Show $T_n(x) = \begin{cases} \cos(n \arccos x), & |x| \le 1 \\ \cosh(n \mathrm{arccosh}\, x), & |x| > 1 \end{cases}$.
        \item Prove $\min_{\text{deg } p \le n, p(1)=1} \|p\|_\infty = 1$, attained at $T_n$.
        \item For Chebyshev nodes $x_j$, prove $\|f - L_n\|_\infty \le \frac{\|f^{(n+1)}\|_\infty}{(n+1)! 2^n}$.
        \item Derive Gauss-Chebyshev quadrature error bound: $|I(f) - I_{n+1}(f)| \le \frac{\pi \|f^{(2n+2)}\|_\infty}{(2n+2)! 2^{2n+1}}$.
    \end{enumerate}

    \item \textbf{(20 points)} Generalized eigenvalue problem $\mu A x = \lambda B x$. UET: $Q^H(A, B)Z = (S, T)$.
    \begin{enumerate}[label=(\alph*)]
        \item Give an example where $(A, B)$ has $> n$ eigenvalues.
        \item Prove existence of UET such that $S, T$ are upper triangular.
        \item Explain how to get eigenvalues from $(S, T)$.
        \item Propose method to construct UET with $A_0$ upper Hessenberg and $B_0$ upper triangular.
    \end{enumerate}

    \item \textbf{(15 points)} Construct Du Fort-Frankel scheme for 2D diffusion $u_t = u_{xx} + u_{yy}$ and discuss consistency and stability.

    \item \textbf{(15 points)} For $u_{tt} = u_{xx}$, analyze stability of:
    \[
    \frac{u_j^{n+1} - 2u_j^n + u_j^{n-1}}{\tau^2} = \frac{u_{j+1}^{n+1} - 2u_j^{n+1} + u_{j-1}^{n+1}}{4h^2} + \frac{u_{j+1}^n - 2u_j^n + u_{j-1}^n}{2h^2} + \frac{u_{j+1}^{n-1} - 2u_j^{n-1} + u_{j-1}^{n-1}}{4h^2}.
    \]

    \item[] \textit{Choose either 6 or 7:}

    \item \textbf{(20 points)} Rayleigh oscillator $y'' + y + \epsilon [\frac{1}{3}(y')^3 - y'] = 0, y(0) = 0, y'(0) = 2a$.
    \begin{enumerate}[label=(\alph*)]
        \item Construct multiple-scale approximation valid for large $t$.
        \item Explain accuracy.
        \item Plot phase plane orbits $y-y'$ for different $a$.
    \end{enumerate}

    \item \textbf{(20 points)} Minimization $\min_{x} \|Ax - b\|_2 + \gamma \|x\|_1$.
    \begin{enumerate}[label=(\alph*)]
        \item Derive Lagrange dual of $\min_{x,y} \|y\|_2 + \gamma \|x\|_1$ s.t. $Ax - b = y$.
        \item For optimal $x^*$, let $r = \frac{Ax^* - b}{\|Ax^* - b\|_2}$. Show $\|A^T r\|_\infty \le \gamma$ and $r^T A x^* + \gamma \|x^*\|_1 = 0$.
        \item Show that if $\|a_i\|_2 < \gamma$, then $x_i^* = 0$.
    \end{enumerate}
\end{enumerate}
