\begin{enumerate}[label=\textbf{\arabic*.}]
    \item \textbf{(10 points)} $f \in C^2(\mathbb{R})$ has unique zero $x_*$ with $f'(x_*) \neq 0$. Consider $x_{k+1} = x_k - \frac{[f(x_k)]^2}{f(x_k + f(x_k)) - f(x_k)}$. Prove $x_k \to x_*$ and estimate local convergence order.

    \item \textbf{(20 points)} $\langle f, g \rangle = \int_{-1}^1 f(x)g(x)dx, \|f\|_2 = \sqrt{\langle f, f \rangle}$. Monic Legendre $P_0(x) = 1, P_1(x) = x, P_{n+1}(x) = x P_n(x) - \frac{n^2}{4n^2-1} P_{n-1}(x)$.
    \begin{enumerate}[label=(\alph*)]
        \item Compute $\langle P_n, P_m \rangle$.
        \item Prove $P_n$ minimizes $\|p\|_2$ among monic polynomials and compute minimal value.
        \item For $f \in C^{n+1}[-1, 1]$, prove interpolation error bound $\|f - L_n\|_2 \le \frac{2 \|f^{(n+1)}\|_2}{(2n+3) a_{n+1} (n+1)!}$.
        \item Derive Gauss-Legendre quadrature $I_{n+1}(f)$ and estimate error $|I(f) - I_{n+1}(f)|$.
    \end{enumerate}

    \item \textbf{(20 points)} Consider $\min_X \|A X B - C\|_F$ ($A \in \mathbb{R}^{l \times m}, B \in \mathbb{R}^{n \times p}, C \in \mathbb{R}^{l \times p}$).
    \begin{enumerate}[label=(\alph*)]
        \item Prove minimal value is 0 iff $\rank(A) = \rank([A \quad C])$ and $\rank(B) = \rank(\begin{bmatrix} B \\ C \end{bmatrix})$.
        \item Show unique minimizer $X_*$ exists iff $\rank(A) = m, \rank(B) = n$.
        \item Propose numerical method to compute unique $X_*$.
        \item Does $X_*$ also minimize $\|A X B - C\|_2$? Why?
    \end{enumerate}

    \item \textbf{(15 points)} For $u_t + u_x = 0$, analyze stability of implicit schemes:
    \begin{enumerate}[label=(\alph*)]
        \item $\frac{u_j^{n+1} - u_j^n}{\tau} + \frac{u_j^{n+1} - u_{j-1}^{n+1}}{h} = 0$
        \item $\frac{u_j^{n+1} - u_j^n}{\tau} + \frac{u_{j+1}^{n+1} - u_j^{n+1}}{h} = 0$
        \item $\frac{u_j^{n+1} - u_j^n}{\tau} + \frac{u_{j+1}^{n+1} - u_{j-1}^{n+1}}{2h} = 0$
    \end{enumerate}

    \item \textbf{(15 points)} Maximum principle of centered finite difference for $u_{xx} + u_{yy} + d u_x + e u_y + f u = 0$ ($f < 0$). Apply to show stability of backward Euler for $u_t - (u_{xx} + u_{yy} + d u_x + e u_y + f u) = 0$.

    \item[] \textit{Choose either 6 or 7:}

    \item \textbf{(20 points)} System $\ddot{x} - 2x - x^2 + x^3 = 0$.
    \begin{enumerate}[label=(\alph*)]
        \item Find static solutions.
        \item For small $\epsilon$, show orbit starting at $(2+\epsilon, 0)$ is closed. Find approximation of period up to $O(\epsilon^2)$.
    \end{enumerate}

    \item \textbf{(20 points)} Polar set $C^\circ = \{y \in \mathbb{R}^n \mid y^T x \le 1, \forall x \in C\}$.
    \begin{enumerate}[label=(\alph*)]
        \item Show $C^\circ$ is convex.
        \item Polar of a cone?
        \item Polar of unit ball for norm $\|\cdot\|$?
        \item Polar of $C = \{x \mid \mathbf{1}^T x = 1, x \succeq 0\}$?
        \item Show $(C^\circ)^\circ = C$ for closed convex $C$ containing 0.
    \end{enumerate}
\end{enumerate}
