\subsection*{Part I: Warm-Up Questions [3 pts.]}
This section contains multiple-choice questions. Please select only one answer for each question. No justification is required.

\begin{enumerate}[label=\textbf{MQ\arabic*.}]
    \item \textbf{[0.5 pts.]} In the agnostic PAC learning setting, what quantity does a learning algorithm aim to compete with?
    \begin{enumerate}[label=(\alph*)]
        \item The Bayes optimal classifier.
        \item The hypothesis with zero training error.
        \item The best hypothesis in the class $\mathcal{H}$.
        \item The hypothesis minimising validation error.
    \end{enumerate}

    \item \textbf{[0.5 pts.]} Let $\mathcal{H}$ be a binary hypothesis class with VC dimension $d < \infty$. Which statement about its growth function $\tau_{\mathcal{H}}(m)$ is necessarily true?
    \begin{enumerate}[label=(\alph*)]
        \item $\tau_{\mathcal{H}}(m) = 2^m$ for all $m$.
        \item $\tau_{\mathcal{H}}(m) = O(m^d)$ for all $m$.
        \item $\tau_{\mathcal{H}}(m) \le \sum_{i=0}^d \binom{m}{i}$ for all $m$.
        \item $\tau_{\mathcal{H}}(m)$ is constant for $m > d$.
    \end{enumerate}

    \item \textbf{[0.5 pts.]} Which statement about ReLU neural networks is correct from a learning-theoretic perspective?
    \begin{enumerate}[label=(\alph*)]
        \item ReLU networks represent smooth functions on $\mathbb{R}^d$.
        \item Increasing depth always decreases the VC dimension.
        \item ReLU networks compute piecewise linear functions whose complexity depends on depth and width.
        \item Universal approximation of ReLU networks implies PAC learnability.
    \end{enumerate}

    \item \textbf{[0.5 pts.]} Which statement about empirical risk minimisation (ERM) is correct?
    \begin{enumerate}[label=(\alph*)]
        \item ERM is guaranteed to be consistent for any hypothesis class.
        \item ERM always finds a hypothesis with minimum true risk.
        \item If $\mathcal{H}$ has finite VC dimension, ERM is PAC learnable in the realisable setting.
        \item ERM is PAC learnable only if $\mathcal{H}$ is finite.
    \end{enumerate}

    \item \textbf{[0.5 pts.]} Which statement is a correct consequence of the No Free Lunch theorem?
    \begin{enumerate}[label=(\alph*)]
        \item Restricting the hypothesis class always improves generalization.
        \item There exists a universally optimal learning algorithm.
        \item Inductive bias is necessary to obtain non-trivial learning guarantees.
        \item Random guessing is optimal for all supervised learning problems.
    \end{enumerate}

    \item \textbf{[0.5 pts.]} Let $K : \mathcal{X} \times \mathcal{X} \to \mathbb{R}$ be a symmetric function. Which of the following statements is correct?
    \begin{enumerate}[label=(\alph*)]
        \item $K$ is a valid kernel if and only if $K(x, x) \ge 0$ for all $x \in \mathcal{X}$.
        \item $K$ is a valid kernel if and only if there exists a finite-dimensional feature map $\psi$ such that $K(x, x') = \langle \psi(x), \psi(x') \rangle$.
        \item If $K$ is positive semidefinite, then there exists a Hilbert space $\mathcal{H}$ and a feature map $\psi : \mathcal{X} \to \mathcal{H}$ such that $K(x, x') = \langle \psi(x), \psi(x') \rangle_{\mathcal{H}}$.
        \item Every kernel corresponds to a unique feature map.
    \end{enumerate}
\end{enumerate}

\subsection*{Part II: Theoretical Exercises [30 pts.]}

\begin{enumerate}[label=\textbf{(\arabic*)}]
    \item \textbf{[6 pts.]} Let $\mathcal{H}$ be a hypothesis class of binary classifiers $h : \mathcal{X} \to \{0, 1\}$. Let $\mathcal{D}$ be an unknown distribution over $\mathcal{X}$, and let $f \in \mathcal{H}$ be the target hypothesis. For a fixed $h \in \mathcal{H}$, define the empirical loss on $S = \{x_1, \dots, x_m\} \sim \mathcal{D}^m$ by
    \[
    L_S(h) = \frac{1}{m} \sum_{i=1}^m \mathbf{1}[h(x_i) \neq f(x_i)].
    \]
    \begin{enumerate}[label=(\alph*)]
        \item Let $p \coloneqq L_{\mathcal{D}, f}(h)$. Show that
        \[
        \E_{S \sim \mathcal{D}^m} \left[ (L_S(h) - p)^2 \right] = \frac{p(1 - p)}{m}.
        \]
        \item Deduce that
        \[
        \E \left[ (L_S(h) - p)^2 \right] \le \frac{1}{4m},
        \]
        and therefore it decreases at rate $O(1/m)$.
    \end{enumerate}

    \item \textbf{[4 pts.]} Let $f : \mathbb{R}^d \to \mathbb{R}$ be twice continuously differentiable and $\lambda$-strongly convex. Prove that $f$ admits a unique global minimiser.

    \item \textbf{[8 pts.]} Let $\mathcal{X} = \mathbb{R}$ and consider the hypothesis class
    \[
    \mathcal{H} = \left\{ h_{a,b,c}(x) = \mathbf{1}_{[a,b]}(x) \lor \mathbf{1}_{[c,\infty)}(x) \mid a \le b < c, \, a, b, c \in \mathbb{R} \right\},
    \]
    where $\lor$ denotes logical OR.
    \begin{enumerate}[label=(\alph*)]
        \item Let $x_1, \dots, x_n$ be $n$ distinct points in $\mathbb{R}$. Give an upper bound on the growth function $\tau(\mathcal{H}, n)$.
        \item Determine the VC dimension of $\mathcal{H}$.
    \end{enumerate}

    \item \textbf{[7 pts.]} Consider a set $S$ of examples in $\mathbb{R}^n \times [k]$ for which there exist vectors $\mu_1, \dots, \mu_k \in \mathbb{R}^n$ such that every example $(x, y) \in S$ falls within a ball centered at $\mu_y$ with radius $r \ge 1$. Assume also that for every $i \neq j$, $\|\mu_i - \mu_j\| \ge 4r$. Concatenate each instance by constant $1$ and apply multivector construction:
    \[
    \Psi(x, y) = \bigl[ \underbrace{0, \dots, 0}_{(y-1)(n+1)}, \underbrace{x_1, \dots, x_n, 1}_{n+1}, \underbrace{0, \dots, 0}_{(k-y)(n+1)} \bigr] \in \mathbb{R}^{k(n+1)}.
    \]
    Show that there exists a vector $w \in \mathbb{R}^{k(n+1)}$ such that $\ell(w, (x, y)) = 0$ for every $(x, y) \in S$.
    \textit{Hint:} Write $x = \mu_y + v$ with $\|v\| \le r$, and take $\mathbf{w} = [\mathbf{w}_1, \dots, \mathbf{w}_k]$ where $\mathbf{w}_i = [\bm{\mu}_i, -\frac{\|\bm{\mu}_i\|^2}{2}]$.

    \item \textbf{[5 pts.]} Let $d \in \mathbb{N}$. Consider two feedforward neural networks with input dimension $d$ and scalar output. Assume $\Phi_{\text{ReLU}}$ uses ReLU activation in all hidden layers, while $\Phi_\sigma$ uses sigmoid activation. Suppose $\Phi_{\text{ReLU}}(x) = \Phi_\sigma(x)$ for all $x \in \mathbb{R}^d$. Show that the function represented by $\Phi_{\text{ReLU}}$ (and hence $\Phi_\sigma$) must be constant.
\end{enumerate}

\section*{C. Optimization Methods in Artificial Intelligence}

Composite optimization: $\min_{x \in \mathbb{R}^d} F(x) \coloneqq f(x) + R(x)$, $f$ is $L$-smooth and $\mu$-strongly convex ($\mu > 0$), $R$ is proper, closed, convex. Unbiased stochastic gradient $g(x, \xi) = \nabla f(x) + \xi$ with $\E[\xi] = 0, \E[\|\xi\|^2] \le \sigma^2$. Proximal SGD: $x^{k+1} = \prox_{\gamma R}(x^k - \gamma g^k)$, $\bar{x}^k \coloneqq \frac{1}{k} \sum_{t=1}^k x^t$.

\begin{enumerate}[label=\textbf{(\arabic*)}]
    \item \textbf{[2 pts.] Basic definitions:} Definitions of $L$-smoothness and $\mu$-strong convexity.
    \item \textbf{[3 pts.] Optimality condition of prox:} Show $x^+ = \prox_{\gamma R}(u) \iff \frac{1}{\gamma}(u - x^+) \in \partial R(x^+)$.
    \item \textbf{[3 pts.] Uniqueness \& fixed point:} Show $F$ has unique minimizer $x^*$ and $x^* = \prox_{\gamma R}(x^* - \gamma \nabla f(x^*))$.
    \item \textbf{[2 pts.] Monotonicity of subdifferential:} Show $\langle r_x - r_y, x - y \rangle \ge 0$ for $r_x \in \partial R(x), r_y \in \partial R(y)$.
    \item \textbf{[3 pts.] Nonexpansiveness of prox:} Show $\|\prox_{\gamma R}(u) - \prox_{\gamma R}(v)\| \le \|u - v\|$.
    \item \textbf{[3 pts.] One-step inequality:} Show $\|x^{k+1} - x^*\|^2 \le \|x^k - x^* - \gamma (g^k - \nabla f(x^*))\|^2$.
    \item \textbf{[5 pts.] Conditional expectation \& variance decomposition:} Prove $\E_k \|x^{k+1} - x^*\|^2 \le \|x^k - x^* - \gamma(\nabla f(x^k) - \nabla f(x^*))\|^2 + \gamma^2 \sigma^2$.
    \item \textbf{[4 pts.] Linear contraction \& noise floor:} For $\gamma = \frac{2}{L + \mu}$, show $\E \|x^{k+1} - x^*\|^2 \le (1 - \rho)\E \|x^k - x^*\|^2 + \gamma^2 \sigma^2$ with $\rho = \frac{4\mu L}{(L+\mu)^2}$. Explain noise floor.
    \item \textbf{[3 pts.] Averaging reduces variance:} For uncorrelated errors $e^t = x^t - \E[x^t]$ with $\E[\|e^t\|^2] \le V$, show $\E[\|\bar{e}^k\|^2] \le V/k$.
    \item \textbf{[5 pts.] Convergence rate of averaging ($R \equiv 0$):}
    \begin{enumerate}[label=(\alph*)]
        \item Show $\E_k \|x^{t+1} - x^*\|^2 \le \|x^t - x^*\|^2 - 2\gamma_t (1 - L\gamma_t) \E_k(f(x^t) - f^*) + \gamma_t^2 \sigma^2$, and for $\gamma_t \le \frac{1}{2L}$, simplify to $\E_k \|x^{t+1} - x^*\|^2 \le \|x^t - x^*\|^2 - \gamma_t \E_k(f(x^t) - f^*) + \gamma_t^2 \sigma^2$.
        \item For $\gamma_t = \frac{1}{\mu t}$, show $\E[f(\bar{x}^k) - f^*] \le \mathcal{O}\left( \frac{\log k}{k} \right)$.
    \end{enumerate}
\end{enumerate}
