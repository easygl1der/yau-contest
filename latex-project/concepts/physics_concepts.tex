\section{Mathematical Physics}
\label{sec:physics_concepts}

The Mathematical Physics track covers analytical mechanics, quantum theory, electromagnetism, statistical mechanics, general relativity, and quantum field theory.

\subsection{Classical Mechanics}

\begin{itemize}
  \item \textbf{Lagrangian and Hamiltonian formalisms}
  \item Polar coordinates and effective 1D reduction
  \item Central potential $V(r)=\alpha r^k$; conserved angular momentum and energy
  \item Circular orbits; effective potential; orbital stability
  \item Small oscillations about equilibrium
  \item Constrained motion; Lagrange multipliers
  \item Rotating hoop dynamics
  \item Noether's theorem; symmetries and conserved currents
  \item Schwinger boson representation of $su(2)$ algebra using two decoupled harmonic oscillators
  \item Central force problem; Binet equation
  \item Bead on rotating hoop: pitchfork bifurcation and spontaneous symmetry breaking
\end{itemize}

\subsection{Quantum Mechanics}

\begin{itemize}
  \item \textbf{Harmonic oscillator}: ladder operators $\hat a$, $\hat a^\dagger$; $su(2)$ algebra; number states
  \item Coherent states; displacement operator
  \item Transition probabilities; Poisson-distributed transitions
  \item Time-dependent perturbation theory
  \item Heisenberg picture; equations of motion
  \item Perturbation theory: degenerate and non-degenerate cases; first-order corrections
  \item Anisotropic harmonic oscillator; exact diagonalization; perturbation correction
  \item Shifted harmonic oscillator; translation operator
  \item Quantum scattering: step potential; barrier potential
  \item Reflection and transmission coefficients
  \item Time-reversal symmetry; anti-unitary operators; Wigner theorem
  \item Aharonov--Bohm magnetic field; gauge-invariant Lagrangian
\end{itemize}

\subsection{Electromagnetism}

\begin{itemize}
  \item Maxwell equations in vacuum and media; constitutive relations
  \item Wave propagation in conductors; complex conductivity
  \item Skin depth and skin effect
  \item Wave reflection at conducting half-space
  \item Scalar and vector potentials; Lorentz gauge
  \item Retarded Green's function; retarded potentials
  \item Li\'enard--Wiechert potentials for a moving point charge
  \item Gaussian wave packet; paraxial approximation
  \item Wave packet propagation; longitudinal field component
  \item Angular momentum of light
\end{itemize}

\subsection{Statistical Mechanics}

\begin{itemize}
  \item Ising model: mean-field theory; Landau free energy
  \item Critical exponents $\alpha, \beta, \gamma$
  \item Ising model on triangular lattice: exact partition function (Onsager)
  \item Specific heat and susceptibility on triangular lattice
  \item Geometric frustration in the triangular Ising model
  \item Free massless bosons; heat capacity; Stefan--Boltzmann law
  \item Fermi--Dirac statistics; Sommerfeld expansion
  \item Degenerate Fermi gas
  \item Partition function; thermodynamic potentials
  \item Magnetization fluctuations
  \item Infinite-range Ising model: mean-field critical exponents
\end{itemize}

\subsection{General Relativity}

\begin{itemize}
  \item Schwarzschild metric; geodesics
  \item Effective potential for geodesic motion
  \item Schwarzschild--de Sitter metric
  \item Vacuum Einstein equations $R_{\mu\nu}=kg_{\mu\nu}$ with cosmological constant
  \item Killing vectors; conserved quantities along geodesics
  \item de Sitter space: static coordinates; geodesic equation
  \item Penrose diagram for de Sitter space
  \item Gravitational waves; binary black hole inspiral
  \item Quadrupole formula; orbital decay
  \item Null hypersurfaces; Eddington--Finkelstein coordinates
  \item Black hole horizons
  \item de Sitter space embedding in higher dimensions; Einstein metric property
\end{itemize}

\subsection{Quantum Field Theory}

\begin{itemize}
  \item $\phi^3$ theory: Feynman diagrams
  \item One-loop self-energy in $\phi^3$ theory
  \item Dimensional regularization
  \item Yukawa theory: scalar and fermion self-energy at one loop
  \item Counterterms
  \item $\phi^4$ theory: loop expansion
  \item Scalar self-energy diagrams in $\phi^4$
  \item $O(N)$ scalar theory; large-$N$ expansion
  \item $\overline{\mathrm{MS}}$ scheme; renormalization conditions
  \item Scale invariance; conformal invariance
  \item Trace of the energy-momentum tensor
  \item Trace anomaly
  \item Conformal scalar field on curved spacetime
  \item Weyl invariance; non-minimal coupling
  \item Curvature-induced effective mass
\end{itemize}

\subsection{Recurring Themes}

\begin{itemize}
  \item \textbf{Symmetries and Conservation Laws:} Noether's theorem; Killing vectors; rotational and scale invariance
  \item \textbf{Effective Potentials and Stability:} Orbital stability; phase transitions; equilibrium analysis
  \item \textbf{Quantum Coherent Structures:} Coherent states; displacement and translation operators
  \item \textbf{Geometric Methods:} Aharonov--Bohm potential; de Sitter embeddings; Eddington--Finkelstein coordinates
  \item \textbf{Renormalization and Scale Dependence:} Regularization; renormalization group; conformal symmetry
\end{itemize}
