\section{Group Theory and Representations}

\begin{itemize}
  \item Group actions, orbits, stabilizers, conjugacy classes
  \item Sylow theorems ($p$-subgroup counting); simple groups
  \item Groups of order 26, 99, 150 (classification/non-simplicity)
  \item $SL_2(\mathbb{C})$ symmetric powers; Clebsch--Gordan decomposition
  \item $SL_2(\mathbb{Z})$ generated by $S=\begin{pmatrix}0&-1\\1&0\end{pmatrix}$ and $T=\begin{pmatrix}1&1\\0&1\end{pmatrix}$
  \item $S_4$ irreducible representations; character table construction
  \item Molien series for invariant rings
  \item Tensor invariants; $O(V)$-invariants in $V\otimes V$; $\mathbb{C}[\mathrm{End}(V)]^{SL_2\times SL_2}$
  \item Faithful representations; cyclic groups; 1-dimensional reps
  \item Representation theory of $SU(2)$ and $SU(3)$ (Young tableaux basics)
  \item Classification of finite subgroups of $GL_2(\mathbb{R})$: $C_n$ or $D_{2n}$
  \item Semi-direct products $\mathbb{Z}/p\mathbb{Z} \ltimes (\mathbb{Z}/q\mathbb{Z})^n$
  \item Goursat's lemma on $S \times T$ normal subgroups
  \item Fitting's lemma and Fitting decomposition for finite abelian groups
  \item Conjugacy class bound $c(G) \leq 5|G|/8$
  \item Rational-valued characters imply $g \sim g^k$ for $(k,|g|)=1$
  \item Character theory: irreducible characters, orthogonality relations
  \item $SL_n(F)$ generated by elementary matrices
  \item Solvability of $GL_n(F)$
\end{itemize}

\section{Galois Theory and Field Extensions}

\begin{itemize}
  \item Field embeddings; $\mathbb{C}\hookrightarrow M_2(\mathbb{R})$, Cayley--Hamilton
  \item Galois groups of polynomials: $x^8-5$ (degree 32, quaternion), $x^5-x+1$ ($\mathrm{Gal}\cong S_5$)
  \item Galois group of $x^4-2$ ($D_4$), $x^3-3x+1$ ($\mathbb{Z}/3\mathbb{Z}$)
  \item Solvable groups; solvable Galois groups; radical extensions
  \item Kummer theory; biquadratic polynomials $X^4+aX^2+b$; $(\mathbb{Z}/2)^2$ extensions
  \item Splitting fields; normal extensions; separability
  \item Inertia groups; ramification index and degree; discriminant
  \item Teichm\"uller character: $\mathbb{Z}_p^\times\cong\mathbb{Z}_p\times\mathbb{Z}/(p-1)$
  \item Semidirect products $C_e\rtimes C_f$; absolutely irreducible $\mathbb{F}_p$-representations
  \item Infinite Galois extensions; cyclic subextensions
  \item Normal basis theorem; trace and norm
  \item Cyclotomic fields $\mathbb{Q}(\zeta_p)$; $\Phi_p$ minimal polynomial
  \item Artin-Schreier and Artin-Schreier-Witt extensions
  \item Galois descent for unramified $p$-adic extensions
  \item Function field extensions $\mathbb{F}_p(t)$; Galois theory of function fields
\end{itemize}
