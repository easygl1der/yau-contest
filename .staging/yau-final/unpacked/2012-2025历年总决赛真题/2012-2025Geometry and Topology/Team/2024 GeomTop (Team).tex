\begin{problem}
Let $K \subset \mathbb{S}^3$ be a smoothly embedded circle (we call
it a \textbf{knot}) in the 3-sphere $\mathbb{S}^3$. Compute all the
homology groups of the complement $\mathbb{S}^3 \setminus K$ over
$\mathbb{Z}$.
\end{problem}


\begin{problem}
\begin{enumerate}
          \item Consider the height function $z$ on the unit sphere $\mathbb{S}^2 \subset \mathbb{R}^3$.  Show that its critical points are non-degenerate (namely, the Hessian matrix at the critical point is non-degenerate).  Moreover, the number of critical points equals to the rank of its total cohomology $H^*(\mathbb{S}^2,\mathbb{R})$.
          \item Let $\pi: \mathbb{S}^3 \to \mathbb{C}\mathbb{P}^1$ be the fibration defined by $[z_1:z_2]$, where $\mathbb{S}^3$ is the unit sphere in $\mathbb{C}^2$.  Consider the pull-back function $\pi^* f$ on $\mathbb{S}^3$.  What are the critical points of $\pi^* f$? Are the critical points non-degenerate? If not, can you perturb $\pi^* f$ so that the resulting function only has non-degenerate critical points?
        \end{enumerate}
\end{problem}


\begin{problem}
Let $n, k$ be positive integers and $2\leq k\leq n-1$. Suppose $S^n$
is the $n$-dimensional unit sphere in $\mathbb R^{n+1}$ given by
\[\{ (x_1, \cdots, x_{n+1})\,|\, \sum_{i=1}^{n+1} x_i^2=1\}.\]
Suppose $H$ is the cone in $\mathbb R^{n+1}$
 given by \[\{(x_1, \cdots, x_{n+1}\,|\,\sum_{i=1}^k x_i^2=c\sum_{i=k+1}^{n+1} x_i^2\},\] for a constant $c>0$.

(1) Show that $H\cap S^n$ is a smooth submanifold of $S^n$.

(2) For what value(s) of  $c$ (in terms of $n$ and $k$) is $H\cap
S^n$ a minimal submanifold of $S^n$?
\end{problem}



\begin{problem}
Let $a$ be a positive constant less than $1$ and
\[f(u)=(1+a^2)\frac{1-u^2}{1+a^2 u^2} \text{ for } u\in (-1, 1).\]
Consider the Riemmanian metric \[f(u)d\phi\otimes
d\phi+\frac{1}{f(u)}du\otimes du\] defined on $\{(u, \phi)\,|\,
-1<u<1, 0<\phi<2\pi\}$.

(1) Show that the metric can be extended to a smooth Riemmanian
metric on $S^2$.

(2) For what value(s) of  $a$  can this metric be realized as the
induced metric on a closed surface in $\mathbb R^3$?
\end{problem}

\begin{problem}
A 2-form $\omega$ on a manifold $X$ is said to be non-degenerate if
the contraction with $\omega$ is an isomorphism $TM \cong T^*M$.
Then a smooth function $H$ on $X$ produces a vector field $V_H$,
which corresponds to $dH$ under the above isomorphism $TM \cong
T^*M$.
        \begin{enumerate}
            \item For example, show that the standard area form on $\mathbb{R}^2$ is non-degenerate.  As an example, let $H = x^2+y^2$.  Compute the vector field $V_H$.  What is the flow generated by this vector field?
            \item Note that the directional derivative $V_H \cdot H$ is zero in the above example.  Prove that this is generally true.
            \item Note that the area form $\omega$ is invariant under the flow in the above example.  Prove that this is generally true, if we further assume the non-degenerate two-form $\omega$ is closed.
        \end{enumerate}
\end{problem}

\begin{problem}\
    \begin{enumerate}
        \item Let $X$ and $Y$ be compact, orientable manifolds of the same dimension. Define the degree of a continuous map $f: X \to Y$.

        \item Show that the degree of any continuos map $f: \mathbb{CP}^m \to \mathbb{CP}^m$ is of the form $k^m$ for some integer $k$.

        \item Conversely, show that for any $k \in \mathbb{Z}_{>0}$, there is a continuous map $f: \mathbb{CP}^m \to \mathbb{CP}^m$ with degree $k^m$.
    \end{enumerate}
\end{problem}

\

\begin{problem}
Show that the matrix multiplication $\vec{z} \cdot \vec{z}^*$, where
$\vec{z}$ denotes a unit column vector in $\mathbb{C}^3$ and
$\vec{z}^*$ is its conjugate transpose row vector, defines a
$U(3)$-equivariant embedding from $\mathbb{C}\mathbb{P}^2$ to the
vector space of $(3\times 3)$ Hermitian matrices.

                Moreover, show that this is an isometric embedding.  Here,
                $\mathbb{C}\mathbb{P}^2$ is equipped with the standard Fubini-Study metric
                $$ \frac{1}{|\vec{z}|^4}\left( |\vec{z}|^2 |d\vec{z}|^2 - (d\vec{z}^* \cdot \vec{z}) (\vec{z}^* \cdot d\vec{z})\right).$$
                The vector space of  Hermitian matrices is equipped with the flat metric.
\end{problem}

\

\begin{problem}
Let $(S^2, g)$ be the two-dimensional unit sphere with the standard
round metric $g$ and $\nabla$ be the covariant derivative associated
with $g$. Suppose $\omega$ is a smooth differential 1-form on $S^2$
such that $\nabla \omega$ is proportional to $g$ at each point.

(1) Find all such $\omega$.

(2) What is the diffeomorphism (of $S^2$) generated by the dual of
$\omega$?
\end{problem}
