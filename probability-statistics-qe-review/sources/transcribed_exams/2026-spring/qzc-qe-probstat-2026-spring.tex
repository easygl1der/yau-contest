\section*{Instructions}
\begin{itemize}
    \item There are 11 problems in this exam. You need to choose 8 of them to solve. If you select more than 8, only the first 8 that you have worked on will be graded. Note that 4 of the problems are worth 15 points each and the rest 10 points each.
    \item You must follow all the rules of exam taking. Misconducts will be subject to proper disciplinary actions by the Center.
    \item You must provide all necessary details for full credits. A final answer with no or little explanation/derivation, even if correct, receives a minimal credit.
    \item $\mathbb{R}$ denotes the set of real numbers and $\mathbb{N} = \{1, 2, 3, \dots\}$ denotes the set of positive integers. $\xrightarrow{(d)}$ and $\stackrel{(d)}{=}$ mean ``converges in distribution'' and ``equal in distribution'', respectively.
    \item The pmf of a $\mathrm{Poisson}(\lambda)$ random variable $X$ is $P(X = x \mid \lambda) = \frac{e^{-\lambda} \lambda^x}{x!}$ for $x = 0, 1, \dots; \lambda > 0$.
\end{itemize}

\noindent\rule{\linewidth}{0.4pt}

\begin{enumerate}[label=\textbf{\arabic*.}]
    \item \textbf{(10 points)} Let $X = (X_1, \dots, X_n)^T$ be an $\mathbb{R}^n$-valued Gaussian random vector under measure $P$, with mean vector $\bm{\mu} \in \mathbb{R}^n$ and covariance matrix $V \in \mathbb{R}^{n \times n}$. Let $\alpha \in \mathbb{R}^n$ and define a new measure $Q$ by
    \[
    \frac{\mathrm{d}\mathbb{Q}}{\mathrm{d}\mathbb{P}} = \frac{\exp(\alpha^T X)}{\E_{\mathbb{P}}[\exp(\alpha^T X)]}.
    \]
    Show that $X$ is still Gaussian under $Q$ and identify the mean and covariance matrix of $X$ under $Q$.

    \item \textbf{(15 points)} Let $\{X_n\}_{n \in \mathbb{N}}$ be independent with distribution
    \[
    \Pbb[X_n = n^2] = \Pbb[X_n = -n^2] = \frac{1}{6n^2}, \quad \Pbb[X_n = n] = \Pbb[X_n = -n] = \frac{1}{6}, \quad \Pbb[X_n = 0] = \frac{2}{3} - \frac{1}{3n^2}.
    \]
    Let $S_n = \sum_{j=1}^n X_j$. Find $\{b_n\}_{n \in \mathbb{N}}$ such that $S_n / b_n \xrightarrow{(d)} \mathcal{N}(0, 1)$.

    \item \textbf{(10 points)} We have 100 noodles in a bowl. Each noodle has two free ends (200 ends total). We repeatedly choose uniformly at random two free ends and connect them together until no free ends remain. Compute the expectation of the total number of disjoint loops formed.

    \item \textbf{(10 points)} Let $\{B_t\}_{t \ge 0}$ be 1D standard Brownian motion with $B_0 = 0$. Find and prove the limit in distribution of
    \[
    \left[ \int_0^{T^2} \exp(B_t) dt \right]^{1/T} \quad \text{as } T \to \infty.
    \]
    Write your answer as a function of the standard normal distribution.

    \item \textbf{(10 points)} A Galton-Watson branching process $\{Z_n\}$ evolves as $Z_0 = 1, Z_{n+1} = \sum_{j=1}^{Z_n} X(j, n)$, where $X(j, n)$ are i.i.d.\ with mean $m = \E[X(1, 1)] < \infty$. Let $p_k = \Pbb[X(1, 1) = k]$ with $p_0 > 0, p_0 + p_1 < 1$. Let $f(s) = \E[s^{X(1, 1)}]$.
    \begin{enumerate}[label=(\alph*)]
        \item Compute $\E[Z_{n+1} \mid \mathcal{F}_n], \E[Z_n], \E[s^{Z_{n+1}} \mid \mathcal{F}_n], \E[s^{Z_n}]$.
        \item Determine the number of solutions of $f(s) = s$ on $[0, 1]$.
        \item Let $\tau \coloneqq \inf\{n \in \mathbb{N} : Z_n = 0\}$. Compute $\Pbb[\tau < \infty]$ as a function of $q$, the smallest solution of $f(s) = s$ on $[0, 1]$.
    \end{enumerate}

    \item \textbf{(15 points)} Consider the Galton-Watson tree $T$. A property of rooted trees is inherited if it holds for all finite trees, and whenever it holds for a tree, it holds for all subtrees rooted at children of the root.
    \begin{enumerate}[label=(\alph*)]
        \item Prove that for a Galton-Watson tree $T$, conditioned on non-extinction, an inherited property $A$ has probability 0 or 1.
        \item Perform percolation on $T$ with parameter $p$ ($m > 1$). Let $\mathcal{C}_0$ be the open cluster containing root. Conditioned on non-extinction of $T$, compute $p_c(T) \coloneqq \sup \{p \in [0, 1] : \Pbb_p(|\mathcal{C}_0| = \infty \mid T) = 0\}$.
    \end{enumerate}

    \item \textbf{(10 points)} Let $X_1, \dots, X_n$ be i.i.d.\ observations with pdf $f(x; \theta) = \frac{1}{2\theta\sqrt{x}} e^{-\sqrt{x}/\theta} I_{\{x > 0\}}$, where $\theta > 0$.
    \begin{enumerate}[label=(\alph*)]
        \item Show that the family has a monotone likelihood ratio in a statistic $T$.
        \item Find a size $\alpha$ UMP test for $H_0 : \theta \ge 1$ vs. $H_1 : \theta < 1/2$.
    \end{enumerate}

    \item \textbf{(15 points)} Let $X_1, \dots, X_n \stackrel{\text{i.i.d.}}{\sim} \mathrm{Bernoulli}(\theta)$ with $0 < \theta < 1$. Let $\hat{\xi}_p = \inf\{x : F_n(x) \ge p\}$ be the sample quantile.
    \begin{enumerate}[label=(\alph*)]
        \item Does $\hat{\xi}_p$ converge in probability to population quantile $\xi_p$ for each $0 < p < 1$ as $n \to \infty$?
        \item Is it possible for a non-degenerate discrete distribution $F$ to have $\hat{\xi}_p \xrightarrow{\Pbb} \xi_p$ for all $0 < p < 1$? Prove your answer.
    \end{enumerate}

    \item \textbf{(10 points)} Single observation $Y \mid \mu \sim \mathcal{N}(\mu, 1)$, prior $\pi(\mu) = 0.5 I(\mu = -1) + 0.5 I(\mu = 1)$.
    \begin{enumerate}[label=(\alph*)]
        \item Write down posterior $\pi(\mu \mid Y = y)$ and derive Bayes rule $\delta$ under absolute error loss $L(\mu, a) = |\mu - a|$.
        \item Is $\delta$ minimax under absolute error loss? Justify.
    \end{enumerate}

    \item \textbf{(10 points)} Let $X_1, \dots, X_n \stackrel{\text{i.i.d.}}{\sim} \mathrm{Poisson}(\sqrt{\lambda})$ for $\lambda > 0$.
    \begin{enumerate}[label=(\alph*)]
        \item Find the UMVUE $\tilde{\lambda}$ for $\lambda$.
        \item Let $\hat{\lambda}$ be the MLE for $\lambda$. Show that $\sqrt{n}(\tilde{\lambda} - \hat{\lambda}) \xrightarrow{\Pbb} 0$ as $n \to \infty$.
    \end{enumerate}

    \item \textbf{(15 points)} Let $Y = \sigma(\rho |U| + \sqrt{1 - \rho^2} V)$ where $U, V \stackrel{\text{i.i.d.}}{\sim} \mathcal{N}(0, 1)$, $\rho \in (-1, 1)$, $\sigma > 0$.
    \begin{enumerate}[label=(\alph*)]
        \item Show the pdf of $Y$ is $f(y \mid \rho, \sigma) = \frac{a}{\sqrt{2\pi\sigma^2}} \exp\left(-\frac{y^2}{2\sigma^2}\right) \Phi\left(\frac{\rho y}{b\sigma\sqrt{1-\rho^2}}\right)$ for constants $a, b$. Find $a, b$.
        \item Does an MLE of $(\rho, \sigma)$ always exist for $n$ observations? Justify.
        \item For testing $H_0 : \rho = 0$ vs. $H_1 : \rho \neq 0$, find $c$ such that the test rejecting $H_0$ if $\frac{\sqrt{n}|\overline{Y}|}{s_Y} > c$ has size $\alpha = 0.05$.
    \end{enumerate}
\end{enumerate}
