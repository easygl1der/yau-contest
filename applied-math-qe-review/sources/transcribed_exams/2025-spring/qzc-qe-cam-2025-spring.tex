\begin{enumerate}[label=\textbf{\arabic*.}]
    \item \textbf{(15 points)} $f$ periodic with period $2\pi$. $I(f) = \int_0^{2\pi} f(x)dx$, composite trapezoidal $I_h^{Tr}(f) = h \sum_{i=1}^n f(ih)$ ($h = 2\pi/n$).
    \begin{enumerate}[label=(\roman*)]
        \item If $f \in C^m([0, 2\pi])$, prove $|I(f) - I_h^{Tr}(f)| \le C \cdot h^m$.
        \item If $f$ is analytic, prove exponential convergence.
    \end{enumerate}

    \item \textbf{(15 points)} $f \in C^m$ ($m \ge 1$), $f(\alpha) = \dots = f^{(m-1)}(\alpha) = 0, f^{(m)}(\alpha) \neq 0$.
    \begin{enumerate}[label=(\roman*)]
        \item If $m = 1$, prove local quadratic convergence of Newton's method.
        \item If $m > 1$, what is rate of Newton? How about modified $x_{k+1} = x_k - m \frac{f(x_k)}{f'(x_k)}$?
        \item If $m > 1$ unknown, propose a superlinearly convergent iteration.
    \end{enumerate}

    \item \textbf{(20 points)} Sherman-Morrison formula: $(A + uv^T)^{-1} = A^{-1} - \frac{A^{-1}uv^TA^{-1}}{1 + v^T A^{-1}u}$.
    \begin{enumerate}[label=(\roman*)]
        \item Prove the statement.
        \item $A_{j,k} = 4n \delta_{j,k} + \cos(t_j - s_k)$. Give $O(n)$ algorithms for $Ab, A^{-1}$, and $|A|$.
    \end{enumerate}

    \item \textbf{(15 points)} For $u_t = -v_{xx}, v_t = u_{xx}$, analyze truncation error and stability of:
    \[
    \frac{u_j^{n+1} - u_j^n}{\tau} = -\frac{1}{2h^2} (v_{j+1}^n - 2v_j^n + v_{j-1}^n + v_{j+1}^{n+1} - 2v_j^{n+1} + v_{j-1}^{n+1}),
    \]
    \[
    \frac{v_j^{n+1} - v_j^n}{\tau} = \frac{1}{2h^2} (u_{j+1}^n - 2u_j^n + u_{j-1}^n + u_{j+1}^{n+1} - 2u_j^{n+1} + u_{j-1}^{n+1}).
    \]

    \item \textbf{(15 points)} Lax-Wendroff for $u_t + a u_x = 0$: $u_j^{n+1} = -\frac{1}{2}\nu(1-\nu)u_{j+1}^n + (1-\nu^2)u_j^n + \frac{1}{2}\nu(1+\nu)u_{j-1}^n$ ($\nu = a\tau/h$).
    \begin{enumerate}[label=(\roman*)]
        \item Prove $\|u^{n+1}\|_2^2 = \|u^n\|_2^2 - \frac{1}{2}\nu^2(1-\nu^2) \left( \|\delta_x^+ u^n\|_2^2 - \langle \delta_x^+ u^n, \delta_x^- u^n \rangle \right)$.
        \item For $a > 0$ on $(0, 1)$ with $u_0^n = 0$, give numerical boundary condition at $x=1$ for stability.
    \end{enumerate}

    \item[] \textit{Choose either 6 or 7:}

    \item \textbf{(20 points)} $y'' + y + \epsilon (y')^3 = 0, y(0) = 1, y'(0) = 0$.
    \begin{enumerate}[label=(\roman*)]
        \item Use multiple-scale method to construct asymptotic solution for large $t$.
        \item Justify validity.
    \end{enumerate}

    \item \textbf{(20 points)} $K$ closed convex. Projection $\Pi_K(z) = \argmin_{y \in K} \frac{1}{2}\|y - z\|^2$.
    \begin{enumerate}[label=(\roman*)]
        \item Show $\langle z - \Pi_K(z), d - \Pi_K(z) \rangle \le 0, \forall d \in K$.
        \item For $\Theta(z) = \frac{1}{2}\|z - \Pi_K(z)\|^2$, prove $\nabla \Theta(z) = z - \Pi_K(z)$.
    \end{enumerate}
\end{enumerate}
