\begin{enumerate}[label=\textbf{(\arabic*)}]
    \item \textbf{(20 points)} $T_n(x) = \cos(n \arccos(x))$.
    \begin{enumerate}[label=(\roman*)]
        \item Prove $T_n(x)$ is a polynomial of degree $n$.
        \item Show $T_n(2x - 1) = T_{2n}(\sqrt{x})$ for $0 < x < 1$.
        \item Prove $\{T_k : 0 \le k < n\}$ is orthogonal w.r.t.\ discrete inner product $(u, v) = \sum_{k=1}^n u(x_k)v(x_k)$ where $x_k = \cos(\frac{2k-1}{2n}\pi)$.
    \end{enumerate}

    \item \textbf{(15 points)} $A$ symmetric positive definite. Conjugate gradient method solves $Ax = b$.
    \begin{enumerate}[label=(\roman*)]
        \item Prove error estimate $\|x_k - x^*\|_A \le \left(\frac{\sqrt{\kappa_2(A)} - 1}{\sqrt{\kappa_2(A)} + 1}\right)^k \|x_0 - x^*\|_A$.
        \item Describe one Krylov subspace method for non-Hermitian matrices ($A \neq A^*$).
    \end{enumerate}

    \item \textbf{(15 points)} Kepler's equation $f(x) = x - \epsilon \sin x - \eta$ ($0 < |\epsilon| < 1$).
    \begin{enumerate}[label=(\roman*)]
        \item Show exactly one real root $\alpha \in [\eta - |\epsilon|, \eta + |\epsilon|]$.
        \item Show fixed point iteration $x_{n+1} = \epsilon \sin x_n + \eta$ converges for any $x_0$.
        \item For $m\pi < \eta < (m+1)\pi$, prove Newton's method converges monotonically for specified $x_0$.
    \end{enumerate}

    \item \textbf{(15 points)} For $u_t = v_x, v_t = u_x$, analyze truncation error and stability of:
    \[
    \frac{1}{\tau} \left(u_j^{n+1} - \frac{1}{2}(u_{j+1}^n + u_{j-1}^n)\right) = \frac{1}{2h}(v_{j+1}^n - v_{j-1}^n),
    \]
    \[
    \frac{1}{\tau} \left(v_j^{n+1} - \frac{1}{2}(v_{j+1}^n + v_{j-1}^n)\right) = \frac{1}{2h}(u_{j+1}^n - u_{j-1}^n).
    \]

    \item \textbf{(15 points)} Write and prove maximum principle of centered finite difference for $u_{xx} + u_{yy} + d(x,y)u_x + e(x,y)u_y + f(x,y)u = 0$ ($f < 0$).

    \item[] \textit{Choose either (6) or (7):}

    \item \textbf{(20 points)} Boundary value problem $\epsilon y'' + \epsilon(1+x)^2 y' - y = x - 1$, $y(0) = \alpha, y(1) = -1$.
    \begin{enumerate}[label=(\roman*)]
        \item For $\alpha = 1$, construct composite expansion and sketch solution.
        \item For $\alpha = 0$, construct composite expansion.
        \item Estimate accuracy in $\epsilon$.
    \end{enumerate}

    \item \textbf{(20 points)} Convex $f : \mathbb{R}^n \to \mathbb{R}$, subgradient set $\partial f(x)$.
    \begin{enumerate}[label=(\roman*)]
        \item For $x \in \mathrm{int\,dom}\,f$, prove $\partial f(x)$ is nonempty and bounded.
        \item For $f = \|x\|_1$, write down $\partial f(x)$.
        \item Solve $\min_x \frac{1}{2}\|x - y\|_2^2 + \lambda \|x\|_1$.
    \end{enumerate}
\end{enumerate}
